/* ============================================================ סטודיו (תיבות בסגנון BandLab) ============================================================ */ const ST = { tracks: [], playing: false, bpm: 100, steps: 16, step: 0, nextT: 0, timer: null, uid: 1 }; const MAXSTEPS = 32; const NOTE_NAMES = ['C','C#','D','D#','E','F','F#','G','G#','A','A#','B']; const midiName = m => NOTE_NAMES[m % 12] + (Math.floor(m / 12) - 1); const TRACK_COLORS = ['#8b5cf6','#22d3ee','#f472b6','#34d399','#fbbf24','#fb7185','#60a5fa']; const r2 = x => Math.round(x * 100) / 100; function mk(tag, cls, txt) { const e = document.createElement(tag); if (cls) e.className = cls; if (txt !== undefined) e.textContent = txt; return e; } function mkSel(opts, val, onch) { const s = mk('select', 'sel'); opts.forEach(o => { const op = document.createElement('option'); op.value = Array.isArray(o) ? o[0] : o; op.textContent = Array.isArray(o) ? o[1] : o; s.appendChild(op); }); s.value = String(val); s.addEventListener('change', () => onch(s.value)); return s; } const rootMidi = t => 12 * (t.oct + 1) + t.pc; function trackRows(t) { return Array.from(scale(rootMidi(t), t.scale, { num_octaves: 2 })).reverse(); } function shiftGrid(t, d) { const ng = {}; Object.keys(t.grid).forEach(m => { ng[+m + d] = t.grid[m]; }); t.grid = ng; } function pruneGrid(t) { const rows = new Set(trackRows(t).map(String)); Object.keys(t.grid).forEach(m => { if (!rows.has(m)) delete t.grid[m]; }); } function setNotes(t, map) { Object.keys(map).forEach(m => { t.grid[m] = Array(MAXSTEPS).fill(false); map[m].forEach(i => { t.grid[m][i] = true; }); }); } function addTrack(kind, inst, extra) { const t = Object.assign({ id: ST.uid++, kind: kind, inst: inst || (kind === 'drum' ? 'bd_haus' : 'prophet'), vol: 0.8, mute: false, solo: false, color: TRACK_COLORS[ST.uid % TRACK_COLORS.length], cells: Array(MAXSTEPS).fill(false), grid: {}, pc: 0, oct: 3, scale: 'minor_pentatonic', len: 1, ph: null }, extra || {}); ST.tracks.push(t); return t; } function preview(t, m) { try { initAudio(); } catch (e) { return; } const c = { vtime: A.ctx.currentTime, bpm: ST.bpm, out: A.bus }; if (t.kind === 'drum') triggerSample(t.inst, { amp: t.vol }, c); else triggerSynth(t.inst, m === undefined ? rootMidi(t) : m, { amp: t.vol, sustain: 0.15, release: 0.25 }, c); } let paint = null; document.addEventListener('pointerup', () => { paint = null; }); document.addEventListener('pointercancel', () => { paint = null; }); function bindCell(el, get, set, prev) { el.addEventListener('pointerdown', e => { e.preventDefault(); try { el.releasePointerCapture(e.pointerId); } catch (x) {} paint = !get(); set(paint); if (paint && prev) prev(); }); el.addEventListener('pointerenter', () => { if (paint !== null && get() !== paint) set(paint); }); } function renderTrack(t) { const el = mk('div', 'track'); el.style.setProperty('--tc', t.color); // ----- כותרת הערוץ ----- const head = mk('div', 'thead'); const r1 = mk('div', 'trow'); r1.appendChild(mk('span', 'tdot')); r1.appendChild(mkSel(t.kind === 'drum' ? Object.keys(SAMPLES) : Object.keys(SYNTHS), t.inst, v => { t.inst = v; preview(t); })); const del = mk('button', 'btn sm', '✕'); del.title = 'מחק ערוץ'; del.addEventListener('click', () => { ST.tracks = ST.tracks.filter(x => x !== t); renderStudio(); }); r1.appendChild(del); head.appendChild(r1); const r2row = mk('div', 'trow'); const vol = document.createElement('input'); vol.type = 'range'; vol.min = 0; vol.max = 1; vol.step = 0.01; vol.value = t.vol; vol.addEventListener('input', () => { t.vol = +vol.value; }); const bm = mk('button', 'btn sm tog', 'M'); bm.title = 'השתק'; bm.classList.toggle('on', t.mute); bm.addEventListener('click', () => { t.mute = !t.mute; bm.classList.toggle('on', t.mute); }); const bs = mk('button', 'btn sm tog s', 'S'); bs.title = 'סולו'; bs.classList.toggle('on', t.solo); bs.addEventListener('click', () => { t.solo = !t.solo; bs.classList.toggle('on', t.solo); }); const cl = mk('button', 'btn sm', 'נקה'); cl.addEventListener('click', () => { t.cells = Array(MAXSTEPS).fill(false); t.grid = {}; renderStudio(); }); r2row.appendChild(vol); r2row.appendChild(bm); r2row.appendChild(bs); r2row.appendChild(cl); head.appendChild(r2row); if (t.kind === 'synth') { const r3 = mk('div', 'trow'); r3.appendChild(mkSel(NOTE_NAMES.map((n, i) => [i, n]), t.pc, v => { const old = rootMidi(t); t.pc = +v; shiftGrid(t, rootMidi(t) - old); renderStudio(); })); r3.appendChild(mkSel([1, 2, 3, 4, 5, 6], t.oct, v => { const old = rootMidi(t); t.oct = +v; shiftGrid(t, rootMidi(t) - old); renderStudio(); })); r3.appendChild(mkSel(Object.keys(SCALES), t.scale, v => { t.scale = v; pruneGrid(t); renderStudio(); })); r3.appendChild(mkSel([[1, 'תו: 1 צעד'], [2, 'תו: 2 צעדים'], [4, 'תו: 4 צעדים']], t.len, v => { t.len = +v; })); head.appendChild(r3); } el.appendChild(head); // ----- תיבת הקליפ ----- const clip = mk('div', 'clip'); const area = mk('div', 'area'); area.style.minWidth = (ST.steps * 24) + 'px'; const cells = mk('div', 'cells ' + (t.kind === 'drum' ? 'drum' : 'syn')); cells.style.gridTemplateColumns = 'repeat(' + ST.steps + ', 1fr)'; if (t.kind === 'drum') { for (let i = 0; i < ST.steps; i++) { const c = mk('div', 'cell' + (((i >> 2) & 1) ? ' alt' : '') + (t.cells[i] ? ' on' : '')); bindCell(c, () => !!t.cells[i], v => { t.cells[i] = v; c.classList.toggle('on', v); }, () => preview(t)); cells.appendChild(c); } } else { const rows = trackRows(t); const keys = mk('div', 'keys'); rows.forEach(m => { keys.appendChild(mk('div', 'key' + ((m - rootMidi(t)) % 12 === 0 ? ' rt' : ''), midiName(m))); for (let i = 0; i < ST.steps; i++) { const on = !!(t.grid[m] && t.grid[m][i]); const c = mk('div', 'cell' + (((i >> 2) & 1) ? ' alt' : '') + (on ? ' on' : '')); bindCell(c, () => !!(t.grid[m] && t.grid[m][i]), v => { if (!t.grid[m]) t.grid[m] = Array(MAXSTEPS).fill(false); t.grid[m][i] = v; c.classList.toggle('on', v); }, () => preview(t, m)); cells.appendChild(c); } }); clip.appendChild(keys); } area.appendChild(cells); const ph = mk('div', 'ph'); area.appendChild(ph); t.ph = ph; clip.appendChild(area); el.appendChild(clip); return el; } function renderStudio() { const box = $('#stTracks'); box.textContent = ''; if (!ST.tracks.length) { box.appendChild(mk('div', 'empty', 'אין ערוצים. הוסף תוף או סינת׳ מהסרגל למעלה.')); return; } ST.tracks.forEach(t => box.appendChild(renderTrack(t))); } function movePlayhead(step) { ST.tracks.forEach(t => { if (!t.ph) return; t.ph.style.display = 'block'; t.ph.style.left = (step * 100 / ST.steps) + '%'; t.ph.style.width = (100 / ST.steps) + '%'; }); } function studioStep(step, t0) { const anySolo = ST.tracks.some(x => x.solo); const c = { vtime: t0, bpm: ST.bpm, out: A.bus }; ST.tracks.forEach(t => { if (t.mute || (anySolo && !t.solo)) return; if (t.kind === 'drum') { if (t.cells[step]) triggerSample(t.inst, { amp: t.vol }, c); } else { Object.keys(t.grid).forEach(m => { if (t.grid[m][step]) { triggerSynth(t.inst, +m, { amp: t.vol, sustain: Math.max(0, t.len * 0.25 - 0.05), release: 0.25 }, c); } }); } }); } function studioTick() { if (!ST.playing) return; const ctx = A.ctx, stepDur = 60 / ST.bpm / 4; if (ST.nextT < ctx.currentTime - 0.1) ST.nextT = ctx.currentTime + 0.05; while (ST.nextT < ctx.currentTime + lookahead()) { const s = ST.step, tt = ST.nextT; studioStep(s, tt); setTimeout(() => { if (ST.playing) movePlayhead(s); }, Math.max(0, (tt - ctx.currentTime) * 1000)); ST.nextT += stepDur; ST.step = (ST.step + 1) % ST.steps; } } function studioStart() { try { initAudio(); } catch (e) { log(e.message, 'err'); return; } if (ST.playing) return; ST.playing = true; ST.step = 0; ST.nextT = A.ctx.currentTime + 0.1; ST.timer = setInterval(studioTick, 25); $('#stPlay').classList.add('on'); studioTick(); log('סטודיו: מנגן', 'ok'); } function studioStop() { if (!ST.playing) return; ST.playing = false; clearInterval(ST.timer); $('#stPlay').classList.remove('on'); ST.tracks.forEach(t => { if (t.ph) t.ph.style.display = 'none'; }); log('סטודיו: נעצר', 'info'); } function studioExport() { const anySolo = ST.tracks.some(x => x.solo); const L = ['// נוצר מהסטודיו של BeatVault', 'use_bpm(' + ST.bpm + ');', '']; const used = {}; ST.tracks.forEach(t => { if (t.mute || (anySolo && !t.solo)) return; used[t.inst] = (used[t.inst] || 0) + 1; const name = t.inst + (used[t.inst] > 1 ? '_' + used[t.inst] : ''); L.push("live_loop('" + name + "', () => {"); if (t.kind === 'drum') { const p = []; for (let i = 0; i < ST.steps; i++) p.push(t.cells[i] ? 1 : 0); L.push(' const p = [' + p.join(', ') + '];'); L.push(" if (p[tick('s') % " + ST.steps + "]) sample('" + t.inst + "', {amp: " + r2(t.vol) + '});'); } else { const p = []; for (let i = 0; i < ST.steps; i++) { const ns = []; Object.keys(t.grid).forEach(m => { if (t.grid[m][i]) ns.push(+m); }); ns.sort((a, b) => a - b); p.push('[' + ns.join(', ') + ']'); } L.push(" use_synth('" + t.inst + "');"); L.push(' const p = [' + p.join(', ') + '];'); L.push(" const n = p[tick('s') % " + ST.steps + '];'); L.push(' if (n.length) play(n, {amp: ' + r2(t.vol) + ', sustain: ' + r2(Math.max(0, t.len * 0.25 - 0.05)) + ', release: 0.25});'); } L.push(' sleep(0.25);'); L.push('});'); L.push(''); }); setCode(L.join('\n')); showTab('editor'); log('הסטודיו יוצא לקוד. לחץ Run כדי לנגן אותו כ-live_loop', 'ok'); } function loadDemo() { ST.tracks = []; const d = (inst, steps, extra) => { const t = addTrack('drum', inst, extra); steps.forEach(i => { t.cells[i] = true; }); return t; }; d('bd_haus', [0, 4, 8, 12]); d('sn_dub', [4, 12]); d('hat_tap', [0, 2, 4, 6, 8, 10, 12, 14], { vol: 0.5 }); const b = addTrack('synth', 'bass', { oct: 2, pc: 0, vol: 0.9, len: 2 }); setNotes(b, { 36: [0, 6, 8, 14], 39: [3], 43: [10] }); const l = addTrack('synth', 'prophet', { oct: 4, pc: 0, vol: 0.6, len: 1 }); setNotes(l, { 67: [0, 8], 70: [3], 72: [6], 63: [10], 65: [12] }); } function studioInit() { $('#stPlay').addEventListener('click', studioStart); $('#stStop').addEventListener('click', studioStop); $('#btnStop').addEventListener('click', studioStop); $('#stBpm').addEventListener('input', e => { const v = +e.target.value; if (v >= 30 && v <= 300) ST.bpm = v; }); $('#stSteps').addEventListener('change', e => { ST.steps = +e.target.value; ST.step = ST.step % ST.steps; renderStudio(); }); $('#stAddDrum').addEventListener('click', () => { addTrack('drum', 'bd_haus'); renderStudio(); }); $('#stAddSynth').addEventListener('click', () => { addTrack('synth', 'prophet'); renderStudio(); }); $('#stClear').addEventListener('click', () => { ST.tracks.forEach(t => { t.cells = Array(MAXSTEPS).fill(false); t.grid = {}; }); renderStudio(); }); $('#stExport').addEventListener('click', studioExport); loadDemo(); renderStudio(); } studioInit();
