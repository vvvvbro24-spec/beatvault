// === BeatVault Audio Engine (Sonic Pi style for JS) === const audioCtx = new (window.AudioContext || window.webkitAudioContext)(); // פונקציית עזר להמתנה (הורסת את ה-sleep של Sonic Pi) const sleep = (ms) => new Promise(resolve => setTimeout(resolve, ms)); // פונקציית play - יוצרת צליל בתדר/תו מסוים function play(note, duration = 0.5) { if (audioCtx.state === 'suspended') audioCtx.resume(); const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain(); // המרת תווים למתקן תדרים (או קבלת תדר ישיר) const freq = typeof note === 'number' ? note : noteToFreq(note); osc.type = 'sine'; // אפשר גם 'square', 'sawtooth', 'triangle' osc.frequency.setValueAtTime(freq, audioCtx.currentTime); // מעטפת צליל (Fade out נקי) gain.gain.setValueAtTime(0.3, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration); osc.connect(gain); gain.connect(audioCtx.destination); osc.start(); osc.stop(audioCtx.currentTime + duration); } // המרת תו (כמו C4, A4) לתדר ברץ function noteToFreq(note) { const notes = { "C4": 261.63, "D4": 293.66, "E4": 329.63, "F4": 349.23, "G4": 392.00, "A4": 440.00, "B4": 493.88, "C5": 523.25 }; return notes[note] || 440; } // פונקציית loop - מריצה לולאה בסגנון live_loop async function liveLoop(name, times, callback) { for (let i = 0; i < times; i++) { await callback(i); } }
