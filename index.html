// === BeatVault Advanced Synth Engine (Sonic Pi Style) === let audioCtx = null; let isPlaying = false; function initAudio() { if (!audioCtx) { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } if (audioCtx.state === 'suspended') { audioCtx.resume(); } } const sleep = (ms) => new Promise(resolve => { if (!isPlaying) return; setTimeout(resolve, ms); }); // המרת תווים לתדרים function noteToFreq(note) { const notes = { "C3": 130.81, "D3": 146.83, "E3": 164.81, "F3": 174.61, "G3": 196.00, "A3": 220.00, "B3": 246.94, "C4": 261.63, "D4": 293.66, "E4": 329.63, "F4": 349.23, "G4": 392.00, "A4": 440.00, "B4": 493.88, "C5": 523.25, "D5": 587.33, "E5": 659.25, "F5": 698.46, "G5": 783.99, "A5": 880.00 }; return notes[note] || (typeof note === 'number' ? note : 440); } // פונקציית סינת' מתקדמת (כמו ב-Sonic Pi) function synth(type, note, options = {}) { if (!isPlaying || !audioCtx) return; const freq = noteToFreq(note); const duration = options.duration || 0.4; const cutoff = options.cutoff || 1500; // פילטר תדרים לעידון הסאונד // יצירת אוסילטור (גל קול) const osc = audioCtx.createOscillator(); const filter = audioCtx.createBiquadFilter(); const gain = audioCtx.createGain(); // בחירת סוג כלי הנגינה (צורת הגל) // 'sawtooth' = סאונד אלקטרוני שמן, 'square' = צליל שחיין/גיימינג, 'triangle' = עגול וחם osc.type = type === 'prophet' ? 'sawtooth' : (type === 'beep' ? 'sine' : 'square'); osc.frequency.setValueAtTime(freq, audioCtx.currentTime); // הגדרת פילטר סאונד (חותך את הצורמות ומביא נפח של כלי אמיתי) filter.type = 'lowpass'; filter.frequency.setValueAtTime(cutoff, audioCtx.currentTime); // מעטפת סאונד חלקית (Envelope) למניעת קליקים ורעשים const now = audioCtx.currentTime; gain.gain.setValueAtTime(0, now); gain.gain.linearRampToValueAtTime(0.25, now + 0.05); // Attack מהיר ונקי gain.gain.exponentialRampToValueAtTime(0.0001, now + duration); // Release טבעי // חיבור השרשרת: אוסילטור -> פילטר -> עוצמה -> רמקולים osc.connect(filter); filter.connect(gain); gain.connect(audioCtx.destination); osc.start(now); osc.stop(now + duration); } // תאימות לפקודה הקודמת ליתר ביטחון function play(note, duration = 0.3) { synth('prophet', note, { duration }); } async function loop(times, callback) { for (let i = 0; i < times && isPlaying; i++) { await callback(i); } }
